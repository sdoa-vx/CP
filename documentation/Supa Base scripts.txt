-- 1. Create the Global SDOA Portfolio Table
CREATE TABLE IF NOT EXISTS public.sdoa_portfolio (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    module_id TEXT NOT NULL,
    type TEXT NOT NULL,
    file_path TEXT NOT NULL,
    source_code TEXT NOT NULL,
    workspace_hash TEXT NOT NULL,
    file_hash TEXT,
    version TEXT,
    timestamp TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()),
    UNIQUE(module_id, workspace_hash)
);
-- 2. Create the Portfolio Telemetry / Usage Table
CREATE TABLE IF NOT EXISTS public.portfolio_usage (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    workspace_hash TEXT NOT NULL,
    primitive_count INT DEFAULT 0,
    workflow_count INT DEFAULT 0,
    schema_count INT DEFAULT 0,
    token_count INT DEFAULT 0,
    engine_count INT DEFAULT 0,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now())
);
-- 3. Create the Innovation Events Table
CREATE TABLE IF NOT EXISTS public.innovation_events (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    workspace_hash TEXT NOT NULL,
    detector TEXT NOT NULL,
    file_path TEXT NOT NULL,
    matches INT DEFAULT 1,
    ast_signature TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now())
);
-- 4. Create the Extraction History Table
CREATE TABLE IF NOT EXISTS public.extraction_history (
  id uuid PRIMARY KEY,
  file text NOT NULL,
  module_path text NOT NULL,
  drift_score numeric DEFAULT 0,
  extraction_score numeric DEFAULT 0,
  payload jsonb NOT NULL,
  created_at timestamptz DEFAULT now()
);
-- 5. Add indexes for faster querying
CREATE INDEX IF NOT EXISTS idx_extraction_history_created_at ON public.extraction_history(created_at DESC);
CREATE INDEX IF NOT EXISTS idx_extraction_history_file ON public.extraction_history(file);
-- 6. Disable RLS (Row Level Security) so the backend can freely read/write
ALTER TABLE public.sdoa_portfolio DISABLE ROW LEVEL SECURITY;
ALTER TABLE public.portfolio_usage DISABLE ROW LEVEL SECURITY;
ALTER TABLE public.innovation_events DISABLE ROW LEVEL SECURITY;
ALTER TABLE public.extraction_history DISABLE ROW LEVEL SECURITY;
