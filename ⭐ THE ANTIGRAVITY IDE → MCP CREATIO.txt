1. MCP server code skeletons
server/src/index.ts

ts
import http from "http";
import { handleProposals } from "./routes/proposals";
import { handleDecision } from "./routes/decision";

const PORT = process.env.PORT || 8080;

const server = http.createServer(async (req, res) => {
  try {
    if (!req.url || !req.method) {
      res.statusCode = 400;
      return res.end("Bad Request");
    }

    if (req.url === "/health" && req.method === "GET") {
      res.statusCode = 200;
      return res.end("OK");
    }

    if (req.url === "/fisp/v1/proposals" && req.method === "POST") {
      return handleProposals(req, res);
    }

    if (req.url.startsWith("/fisp/v1/proposals/") && req.method === "POST") {
      return handleDecision(req, res);
    }

    res.statusCode = 404;
    res.end("Not Found");
  } catch (err) {
    console.error("[MCP] Unhandled error:", err);
    res.statusCode = 500;
    res.end("Internal Server Error");
  }
});

server.listen(PORT, () => {
  console.log(`[MCP] Server listening on port ${PORT}`);
});
server/src/routes/proposals.ts

ts
import { IncomingMessage, ServerResponse } from "http";
import { parseJsonBody } from "../utils/parseJsonBody";
import { runProbationOfficer } from "../validators/probationOfficer";
import { checkSemanticSimilarity } from "../fisp/semanticSimilarity";
import { storeProposal } from "../fisp/storeProposal";

export async function handleProposals(req: IncomingMessage, res: ServerResponse) {
  const body = await parseJsonBody(req);

  // Pre-gate: Probation Officer
  const probationResult = runProbationOfficer(body);
  if (!probationResult.ok) {
    res.statusCode = 400;
    return res.end(JSON.stringify({ error: probationResult.reason }));
  }

  // Semantic similarity
  const similarityResult = await checkSemanticSimilarity(body);
  if (similarityResult.merged) {
    res.statusCode = 202;
    return res.end(JSON.stringify({ status: "merged", id: similarityResult.id }));
  }

  const saved = await storeProposal(body);
  res.statusCode = 201;
  res.end(JSON.stringify({ status: "created", id: saved.id }));
}
server/src/routes/decision.ts

ts
import { IncomingMessage, ServerResponse } from "http";
import { parseJsonBody } from "../utils/parseJsonBody";
import { loadProposal } from "../fisp/loadProposal";
import { runCreationPipeline } from "../pipeline/CreationPipeline";

export async function handleDecision(req: IncomingMessage, res: ServerResponse) {
  const body = await parseJsonBody(req);
  const applyToCanonical = !!body.applyToCanonical;

  const id = req.url!.split("/").pop()!;
  const proposal = await loadProposal(id);

  if (!proposal) {
    res.statusCode = 404;
    return res.end(JSON.stringify({ error: "Proposal not found" }));
  }

  if (!applyToCanonical) {
    // mark as rejected, etc.
    res.statusCode = 200;
    return res.end(JSON.stringify({ status: "rejected" }));
  }

  const result = await runCreationPipeline(proposal);

  res.statusCode = result.ok ? 200 : 400;
  res.end(JSON.stringify(result));
}

ts
import { validateLayer, validateType, validateRole } from "../validators";
import { planCanonicalPath } from "./planCanonicalPath";
import { writeCanonicalFile } from "./writeCanonicalFile";
import { runComplianceSuite } from "./runComplianceSuite";
import { openPrForProposal } from "../workers/prWorker";

export interface Proposal {
  id: string;
  type: string;
  name: string;
  version: string;
  source: {
    language: string;
    content: string;
    path: string;
  };
  sdoa: {
    layer: 1 | 2 | 3;
    placement: string;
    manifest: {
      operationalRole: string;
      optimization: {
        priority: string;
        assertionSuite?: string;
      };
    };
  };
  metrics: {
    usageCount: number;
    projectsObserved: number;
    confidence: number;
  };
}

export interface PipelineResult {
  ok: boolean;
  errors?: string[];
  prUrl?: string;
}

export async function runCreationPipeline(proposal: Proposal): Promise<PipelineResult> {
  const errors: string[] = [];

  // Stage 1–2: Perception & Interpretation
  const { sdoa, type, name, source } = proposal;

  // Stage 3: Innovation Detection (server-side sanity)
  if (!source?.content || source.content.trim().length === 0) {
    errors.push("Empty source content");
  }

  // Stage 4: Reuse Mapping (stubbed for now)
  // TODO: check existing canonical modules for overlap

  // Stage 5: Planning
  const plannedPath = planCanonicalPath(proposal);
  if (!plannedPath) {
    errors.push("Unable to determine canonical path");
  }

  // Stage 7: Compliance (we validate before writing)
  if (!validateLayer(sdoa.layer)) errors.push("Invalid SDOA layer");
  if (!validateType(type)) errors.push("Invalid SDOA type");
  if (!validateRole(sdoa.manifest.operationalRole)) errors.push("Invalid operational role");

  if (errors.length > 0) {
    return { ok: false, errors };
  }

  // Stage 6: Creation (file content with header)
  const header = buildAuditHeader(proposal);
  const fileContent = `${header}\n${source.content}`;

  // Stage 8: Migration (write to canonical repo)
  await writeCanonicalFile(plannedPath, fileContent);

  // Stage 9: Verification (tests, lint, etc.)
  const compliance = await runComplianceSuite(plannedPath, proposal);
  if (!compliance.ok) {
    return { ok: false, errors: compliance.errors };
  }

  // Open PR
  const prUrl = await openPrForProposal(proposal, plannedPath);

  return { ok: true, prUrl };
}

function buildAuditHeader(proposal: Proposal): string {
  const now = new Date().toISOString();
  const ext = inferExtensionFromType(proposal.type, proposal.source.language);

  return [
    "// ------------------------------------------------------------------",
    `// File:    ${proposal.name}.${ext}`,
    "// Version: 1.0.0",
    `// Updated: ${now}`,
    `// Changes: Community contribution via FISP Proposal ${proposal.id}`,
    "// ------------------------------------------------------------------",
  ].join("\n");
}

function inferExtensionFromType(type: string, language: string): string {
  if (language === "ts") return "ts";
  if (language === "js") return "js";
  if (language === "json") return "json";
  if (language === "yaml") return "yaml";
  // extend as needed
  return language || "txt";
}

ts
import * as vscode from "vscode";
import { detectInnovation } from "./detectors/innovationDetector";
import { showInnovationPrompt } from "./ui/prompt";
import { submitProposal } from "./api/submitProposal";
import { saveLocalInnovation } from "./storage/localStore";
import { excludeFromFutureChecks } from "./storage/exclusions";

export function activate(context: vscode.ExtensionContext) {
  console.log("[SDOA MCP] VSX extension activated");

  const disposable = vscode.workspace.onDidSaveTextDocument(async (doc) => {
    try {
      const innovation = await detectInnovation(doc);
      if (!innovation) return;

      const choice = await showInnovationPrompt(innovation);

      if (choice === "local") {
        await saveLocalInnovation(innovation);
        vscode.window.showInformationMessage("Saved innovation locally.");
      }

      if (choice === "submit") {
        const result = await submitProposal(innovation);
        vscode.window.showInformationMessage(
          `Submitted to FISP (proposal id: ${result.id || "pending"})`
        );
      }

      if (choice === "exclude") {
        await excludeFromFutureChecks(innovation);
        vscode.window.showInformationMessage("Module excluded from future portfolio checks.");
      }
    } catch (err) {
      console.error("[SDOA MCP] Error in onDidSaveTextDocument:", err);
    }
  });

  context.subscriptions.push(disposable);
}

export function deactivate() {
  console.log("[SDOA MCP] VSX extension deactivated");
}

import * as vscode from "vscode";

export interface InnovationSummary {
  name: string;
  type: string;
  layer: number;
  confidence: number;
  usageCount: number;
}

export async function showInnovationPrompt(
  innovation: InnovationSummary
): Promise<"local" | "submit" | "exclude" | undefined> {
  const msg = [
    "⭐ SDOA Portfolio Expansion Opportunity",
    "",
    `The SDOA Innovation Detector has identified a reusable pattern: ${innovation.name}`,
    `Type: ${innovation.type} (Layer ${innovation.layer})`,
    `Heuristic Confidence: ${Math.round(innovation.confidence * 100)}%`,
    `Occurrences: ${innovation.usageCount}`,
    "",
    "This module is eligible for voluntary contribution to the global SDOA Asset Portfolio.",
  ].join("\n");

  const choice = await vscode.window.showInformationMessage(
    msg,
    { modal: true },
    "Save locally only",
    "Submit to FISP",
    "Exclude from future checks"
  );

  if (choice === "Save locally only") return "local";
  if (choice === "Submit to FISP") return "submit";
  if (choice === "Exclude from future checks") return "exclude";
}

import * as vscode from "vscode";
import { InnovationPayload } from "../types/shared";

export interface SubmitResult {
  id?: string;
  status: "created" | "merged" | "error";
  error?: string;
}

export async function submitProposal(
  innovation: InnovationPayload
): Promise<SubmitResult> {
  try {
    const endpoint = vscode.workspace
      .getConfiguration("sdoaMcp")
      .get<string>("fispEndpoint");

    if (!endpoint) {
      throw new Error("Missing MCP FISP endpoint in settings.");
    }

    const response = await fetch(`${endpoint}/fisp/v1/proposals`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(innovation),
    });

    const json = await response.json();

    if (!response.ok) {
      return { status: "error", error: json.error || "Unknown error" };
    }

    return {
      id: json.id,
      status: json.status === "merged" ? "merged" : "created",
    };
  } catch (err: any) {
    return { status: "error", error: err.message };
  }
}

ts
export function runProbationOfficer(payload: any) {
  const content = payload?.source?.content || "";

  if (content.split("\n").length > 500) {
    return { ok: false, reason: "Line count exceeds 500-line limit" };
  }

  const banned = [/window\./, /document\./, /localStorage/, /sessionStorage/, /eval\(/];

  for (const rule of banned) {
    if (rule.test(content)) {
      return { ok: false, reason: `Prohibited pattern detected: ${rule}` };
    }
  }

  return { ok: true };
}
